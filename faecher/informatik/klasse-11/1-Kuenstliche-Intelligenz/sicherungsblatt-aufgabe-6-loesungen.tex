\begin{methodbox}{blue}{lightblue}{Klasse Perzeptron \(\cdot\) Klassifizieren\\
  \texttt{public int punktKlassifizieren(double x\_1, double x\_2)}}
\begin{lstlisting}[style=java-sheet]
public int punktKlassifizieren(double x_1, double x_2) {
    double gewichtete_Summe = w_1 * x_1 + w_2 * x_2;
    int berechnete_ausgabe = 0;
    if (gewichtete_Summe > theta) {
        berechnete_ausgabe = 1;
    } else {
        berechnete_ausgabe = 0;
    }
    return berechnete_ausgabe;
}
\end{lstlisting}
\end{methodbox}
{\small Die Methode berechnet aus den Eingaben die Ausgabe \textbf{0 oder 1} und gibt sie zurück.
Gewichte und Schwellenwert bleiben dabei unverändert.\par}

\begin{methodbox}{green}{safebg}{Klasse Perzeptron \(\cdot\) Trainieren\\
  \texttt{public void trainieren(Datenpunkt punkt)}\\[0.5mm]
  \normalfont\footnotesize\colorbox{green!18}{\rule{0pt}{1.5mm}\hspace{3mm}}\;Hinterlegt: in der einfachen Variante selbst ergänzt.}
\begin{lstlisting}[style=java-sheet,numbers=left,firstnumber=1,numberstyle=\studentlinemark,numbersep=0pt]
public void trainieren(Datenpunkt punkt) {
    double gewichtete_Summe =
        w_1 * punkt.gibEingabewertX1()
      + w_2 * punkt.gibEingabewertX2();
    int berechnete_ausgabe = 0;
    if (gewichtete_Summe > theta) {
        berechnete_ausgabe = 1;
    } else {
        berechnete_ausgabe = 0;
    }

    int delta = punkt.gibLabel() - berechnete_ausgabe;
    if (delta == 1) {
        w_1 = w_1 + lernrate * punkt.gibEingabewertX1();
        w_2 = w_2 + lernrate * punkt.gibEingabewertX2();
        theta = theta - lernrate;
    } else if (delta == -1) {
        w_1 = w_1 - lernrate * punkt.gibEingabewertX1();
        w_2 = w_2 - lernrate * punkt.gibEingabewertX2();
        theta = theta + lernrate;
    }
}
\end{lstlisting}
\end{methodbox}
{\small Die Methode vergleicht die berechnete Ausgabe mit dem \textbf{Label} des Datenpunkts.
Bei einem Fehler passt sie Gewichte und Schwellenwert an; sie gibt keinen Wert zurück.\par}

\begin{merke}[title=Anpassung im Fehlerfall]
\small
\textbf{Entscheidung:} Nur bei \texttt{gewichtete\_Summe > theta} ist die Ausgabe 1;
sonst ist sie 0, auch bei Gleichheit.\par
\textbf{Fehler:} \texttt{delta = Label - Ausgabe}.
Bei \texttt{delta = 0} bleiben \texttt{w\_1}, \texttt{w\_2} und \texttt{theta} unverändert.\par
\textbf{Parameteränderung:} Gewichtsänderung $=\delta\cdot\text{Lernrate}\cdot\text{Eingabewert}$;\quad
Änderung von $\theta=-\delta\cdot\text{Lernrate}$.\par
\textbf{Datentypen:} \texttt{double}: Eingaben, Gewichte, Lernrate, Summe;\\
\texttt{int}: Label, Ausgabe, \texttt{delta}.
\end{merke}
