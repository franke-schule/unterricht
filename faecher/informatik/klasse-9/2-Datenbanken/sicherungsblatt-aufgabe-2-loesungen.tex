\sectiontitle{2 \textperiodcentered{} Mehrere oder alle Spalten}
\begin{minipage}[t]{.54\linewidth}
\sqlbox{\sqlkw{SELECT} username, birthday\par\sqlkw{FROM} users}
\smallcaption{Ausschnitt der Ergebnisrelation}
\tworesult{username}{birthday}{zornigerfotograf}{2007-10-02}{wanderwut\_finn}{2006-06-14}
\end{minipage}\hfill
\begin{minipage}[t]{.42\linewidth}
\sqlbox{\sqlkw{SELECT} *\par\sqlkw{FROM} users}
\end{minipage}
\merkebox{Die Auswahl von Spalten heißt Projektion. Mehrere Spalten trennt ein Komma; SELECT * gibt alle Spalten aus. Eine Abfrage verändert users nicht.}
\sectiontitle{3 \textperiodcentered{} Sortieren mit ORDER BY}
\begin{minipage}[t]{.43\linewidth}
\sqlbox{\sqlkw{SELECT} username, centimeters\par\sqlkw{FROM} users\par\sqlkw{ORDER BY} centimeters DESC}
\end{minipage}\hfill
\begin{minipage}[t]{.53\linewidth}
\smallcaption{Schematische Sortierung desselben Dreier-Ausschnitts}
{\small\begin{tabularx}{\linewidth}{@{}>{\ttfamily}X>{\ttfamily}r@{\quad}>{\ttfamily}X>{\ttfamily}r@{}}\textbf{vorher}&&\textbf{nachher}&\\
zornigerfotograf&168&wanderwut\_finn&183\\wanderwut\_finn&183&bergcoder&174\\bergcoder&174&zornigerfotograf&168
\end{tabularx}}
\end{minipage}
\vspace{1mm}
DESC: groß nach klein. ASC: klein nach groß; ohne Angabe gilt ASC.

\sectiontitle{4 \textperiodcentered{} Dopplungen vermeiden mit DISTINCT}
\begin{minipage}[t]{.48\linewidth}
\sqlbox{\sqlkw{SELECT} gender\par\sqlkw{FROM} users}
\smallcaption{317 Zeilen; \ldots ist ein abgekürzter Ausschnitt}
\miniresult{gender}{male\\male\\male\\\ldots}
\end{minipage}\hfill
\begin{minipage}[t]{.48\linewidth}
\sqlbox{\sqlkw{SELECT} \sqlkw{DISTINCT} gender\par\sqlkw{FROM} users}
\smallcaption{4 Zeilen}
\miniresult{gender}{male\\female\\diverse\\NULL}
\end{minipage}
\merkebox{DISTINCT steht direkt hinter SELECT und gibt jeden Wert nur einmal aus. Es verändert nur die Ergebnisrelation, nicht die Tabelle users. NULL bedeutet: kein Wert eingetragen.}
\small\color{Muted}NULL rechts ist das Konto admin.\color{Ink}\normalsize\par\vspace{2mm}
\sectiontitle{5 \textperiodcentered{} Zeilen begrenzen mit LIMIT}
\begin{minipage}[t]{.42\linewidth}
\sqlbox{\sqlkw{SELECT} username\par\sqlkw{FROM} users\par\sqlkw{LIMIT} 3}
\smallcaption{Ausschnitt der Ergebnisrelation}
\miniresult{username}{zornigerfotograf\\wanderwut\_finn\\bergcoder}
\par
\vspace{1mm}
LIMIT 3 zeigt höchstens drei Zeilen. Welche drei es sind, hängt von der Reihenfolge ab.
\end{minipage}\hspace{3mm}
\begin{minipage}[t]{.51\linewidth}
\sqlbox{\sqlkw{SELECT} username, birthday\par\sqlkw{FROM} users\par\sqlkw{ORDER BY} birthday DESC\par\sqlkw{LIMIT} 4}
\smallcaption{Ausschnitt der Ergebnisrelation}
{\fontsize{8.5}{10.4}\selectfont
\begin{tabular}{|>{\ttfamily\bfseries}l|>{\ttfamily}l|}\hline
\rowcolor{Sky}username & birthday\\\hline
gartenlesemaus & 2010-07-07\\\hline
gartencoderin & 2010-06-28\\\hline
codekickemmi & 2010-06-22\\\hline
modereiselio & 2010-06-15\\\hline
\end{tabular}}
\par
\vspace{1mm}
Erst stehen die spätesten Geburtsdaten oben; dann bleiben die ersten vier Zeilen übrig.
\end{minipage}
\vspace{2mm}
\merkebox{Reihenfolge: SELECT [DISTINCT] $\rightarrow$ FROM $\rightarrow$ ORDER BY $\rightarrow$ LIMIT. SELECT wählt Spalten, FROM die Tabelle, ORDER BY sortiert und LIMIT begrenzt die Ausgabe.}
